# Boss fight: The Impostor (Inheritance Grove). Written for the campaign.
# The starter's Shape has no virtual functions, so every shape called through
# a Shape* answers with Shape's area of 0, and deleting through a Shape* only
# runs ~Shape. The last test uses a class of its own to see which destructor
# runs, so area stays an ordinary (not pure) virtual function.

TITLE: Virtual Shapes
FUNCTION: virtual area()
GROUP: Boss fights
DIFFICULTY: Medium
TOPIC: poly
MINUTES: 15
PROTOTYPE: virtual double area();
TEXT: Rect and Square both write their own area(), but totalArea calls it through Shape pointers, and every shape answers 0: Shape's version runs every time.
TEXT: Fix Shape so a call through a Shape* runs the right class's area(), and so deleting a shape through a Shape* runs the right destructor too. Keep Shape's own area() returning 0.
RULE: Change Shape, not totalArea
HINT: A call through a base class pointer only looks at the real object's class if the function is virtual.
HINT: virtual double area() { return 0; } in Shape. And any class with virtual functions needs a virtual destructor: virtual ~Shape() {}

EXAMPLE:
Shape *s = new Rect(2, 3);
s->area();                // 6, not 0
Shape *all[] = { new Rect(2, 3), new Square(4) };
totalArea(all, 2);        // 22
END

STARTER:
#include <stdio.h>

class Shape
{
public:
    // your code here
    double area() { return 0; }
    ~Shape() {}
};

class Rect : public Shape
{
public:
    double w, h;
    Rect(double width, double height) : w(width), h(height) {}
    double area() { return w * h; }
};

class Square : public Shape
{
public:
    double side;
    Square(double s) : side(s) {}
    double area() { return side * side; }
};

double totalArea(Shape *shapes[], int n)
{
    double total = 0;
    for (int i = 0; i < n; i++)
        total += shapes[i]->area();
    return total;
}
END

TESTS:
    Shape *r = new Rect(2, 3);
    checkInt((int)r->area(), 6, "a Rect called through a Shape* gives its own area");
    Shape *q = new Square(4);
    checkInt((int)q->area(), 16, "a Square called through a Shape* gives its own area");
    Shape *all[] = { r, q, new Rect(1, 5) };
    checkInt((int)totalArea(all, 3), 27, "totalArea adds up every shape's real area");
    Shape plain;
    checkInt((int)plain.area(), 0, "a plain Shape still has an area of 0");
    static bool gone = false;
    struct Tracker : Shape {
        double area() { return 1; }
        ~Tracker() { gone = true; }
    };
    Shape *t = new Tracker();
    delete t;
    checkBool(gone, true, "deleting through a Shape* runs the derived destructor too");
END
