PROTOTYPE: int ReplaceWithMinusOne(int array[4][6]);
TEXT: Scan the 4 x 6 grid from (0,0), left to right along each row. At each position, look for a horizontal line of 3 or more identical values first; if there isn't one, look for a vertical line.
TEXT: Score the first cluster found by adding up its values (1 to 4), replace it with -1s and return the score. Return 0 if there is no cluster. 0s and -2s are never part of a cluster.
HINT: Write a helper that measures how long a run is from (r, c) going right, and another going down.
HINT: Stop at the first cluster: return straight away once you have replaced it.

EXAMPLE:
{0,0,1,3,3,2},
{0,0,1,1,1,1}   // bottom row: four 1s, score 4
{0,0,-1,-1,-1,-1}   // replaced
END

TESTS:
    // Test 1: the lecturer's grid - the 0 columns must be ignored,
    //         the first real cluster is the four 1's on the bottom row.
    {
        int grid[4][6]     = { {0,0,3,1,3,4},
                               {0,0,2,3,4,3},
                               {0,0,1,3,3,2},
                               {0,0,1,1,1,1} };
        int expected[4][6] = { {0,0,3,1,3,4},
                               {0,0,2,3,4,3},
                               {0,0,1,3,3,2},
                               {0,0,-1,-1,-1,-1} };
        checkInt(ReplaceWithMinusOne(grid), 4, "lecturer's grid: score = 1+1+1+1 = 4");
        checkGrid(grid, expected, "lecturer's grid: bottom-row 1's replaced (0 columns ignored)");
    }

    // Test 2: a vertical line of three 1's in the first column
    {
        int grid[4][6]     = { {1,2,3,4,1,2},
                               {1,3,4,1,2,3},
                               {1,4,1,2,3,4},
                               {2,1,2,3,4,1} };
        int expected[4][6] = { {-1,2,3,4,1,2},
                               {-1,3,4,1,2,3},
                               {-1,4,1,2,3,4},
                               { 2,1,2,3,4,1} };
        checkInt(ReplaceWithMinusOne(grid), 3, "vertical cluster: score = 1+1+1 = 3");
        checkGrid(grid, expected, "vertical cluster replaced");
    }

    // Test 3: horizontal AND vertical both start at (0,0) - horizontal wins
    {
        int grid[4][6]     = { {2,2,2,1,3,4},
                               {2,3,4,1,3,4},
                               {2,4,1,3,1,3},
                               {1,3,4,4,1,2} };
        int expected[4][6] = { {-1,-1,-1,1,3,4},
                               { 2, 3, 4,1,3,4},
                               { 2, 4, 1,3,1,3},
                               { 1, 3, 4,4,1,2} };
        checkInt(ReplaceWithMinusOne(grid), 6, "horizontal checked first: score = 2+2+2 = 6");
        checkGrid(grid, expected, "only the horizontal line replaced, vertical 2's untouched");
    }

    // Test 4: a vertical cluster found earlier in the scan beats a
    //         horizontal cluster in a later row
    {
        int grid[4][6]     = { {0,0,0,0,4,1},
                               {3,3,3,0,4,2},
                               {0,0,0,0,4,3},
                               {0,0,0,0,1,2} };
        int expected[4][6] = { {0,0,0,0,-1,1},
                               {3,3,3,0,-1,2},
                               {0,0,0,0,-1,3},
                               {0,0,0,0, 1,2} };
        checkInt(ReplaceWithMinusOne(grid), 12, "scan order: vertical 4's at (0,4) found before row 1's 3's, score = 12");
        checkGrid(grid, expected, "only the vertical 4's replaced");
    }

    // Test 5: no clusters anywhere
    {
        int grid[4][6]     = { {1,2,3,4,1,2},
                               {2,3,4,1,2,3},
                               {3,4,1,2,3,4},
                               {4,1,2,3,4,1} };
        int expected[4][6] = { {1,2,3,4,1,2},
                               {2,3,4,1,2,3},
                               {3,4,1,2,3,4},
                               {4,1,2,3,4,1} };
        checkInt(ReplaceWithMinusOne(grid), 0, "no cluster: score = 0");
        checkGrid(grid, expected, "no cluster: grid unchanged");
    }

    // Test 6: lines of 0's and -2's must NOT count as clusters
    {
        int grid[4][6]     = { {0,0,0,-2,-2,-2},
                               {1,2,1, 2, 1, 2},
                               {2,1,2, 1, 2, 1},
                               {0,-2,0,-2, 0,-2} };
        int expected[4][6] = { {0,0,0,-2,-2,-2},
                               {1,2,1, 2, 1, 2},
                               {2,1,2, 1, 2, 1},
                               {0,-2,0,-2, 0,-2} };
        checkInt(ReplaceWithMinusOne(grid), 0, "0's and -2's ignored: score = 0");
        checkGrid(grid, expected, "0's and -2's ignored: grid unchanged");
    }
