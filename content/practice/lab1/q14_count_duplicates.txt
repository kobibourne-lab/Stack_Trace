TESTS:
    int a[] = {5, 1, 1};
    checkInt(countDuplicates(3, a), 1, "{5,1,1} -> 1");

    int b[] = {1, 2, 3, 4};
    checkInt(countDuplicates(4, b), 0, "{1,2,3,4} -> 0 (no duplicates)");

    int c[] = {1, 1, 1};
    checkInt(countDuplicates(3, c), 2, "{1,1,1} -> 2");

    int d[] = {1, 2, 1, 2, 3};
    checkInt(countDuplicates(5, d), 2, "{1,2,1,2,3} -> 2 (one extra 1, one extra 2)");

    int e[] = {7, 7, 7, 7};
    checkInt(countDuplicates(4, e), 3, "{7,7,7,7} -> 3");

    int f[] = {3, 1, 3, 1, 3};
    checkInt(countDuplicates(5, f), 3, "{3,1,3,1,3} -> 3 (two extra 3's, one extra 1)");

    int g[] = {9};
    checkInt(countDuplicates(1, g), 0, "{9} -> 0");

    int empty[1] = {0};
    checkInt(countDuplicates(0, empty), 0, "empty array -> 0");
END
