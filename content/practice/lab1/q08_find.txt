    int arr[] = {5, 3, 8, 1, 9};
    checkInt(find(5, arr, 8),  2, "find 8 in {5,3,8,1,9} -> index 2");

    checkInt(find(0, empty, 0), -1, "find anything in an empty array (size 0) -> -1");
