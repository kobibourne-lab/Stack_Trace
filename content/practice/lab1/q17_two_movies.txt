    int a[] = {90, 85, 95, 120};
    checkBool(twoMovies(180, a, 4), true, "180 with {90,85,95,120} -> true (85 + 95)");

    checkBool(twoMovies(180, f, 2), true, "180 with {90,90} -> true (two DIFFERENT movies that are both 90)");
