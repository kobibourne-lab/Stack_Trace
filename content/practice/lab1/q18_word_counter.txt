TESTS:
    #define LEN(s) ((int)sizeof(s) - 1)
    char a[] = "Hello world";
    checkInt(wordCounter(LEN(a), a), 2, "\"Hello world\" -> 2");

    char b[] = "one";
    checkInt(wordCounter(LEN(b), b), 1, "\"one\" -> 1");

    char c[] = "a b c";
    checkInt(wordCounter(LEN(c), c), 3, "\"a b c\" -> 3 (single-letter words)");

    char d[] = "The quick brown fox";
    checkInt(wordCounter(LEN(d), d), 4, "\"The quick brown fox\" -> 4");

    char e[] = "  Hello   world  ";
    checkInt(wordCounter(LEN(e), e), 2, "\"  Hello   world  \" -> 2 (extra spaces everywhere)");

    char f[] = "     ";
    checkInt(wordCounter(LEN(f), f), 0, "\"     \" -> 0 (only spaces)");

    char g[] = "";
    checkInt(wordCounter(LEN(g), g), 0, "\"\" -> 0 (empty)");
